### Clearable with Event

<CodeExample
  html={`<form class="form">
  <m-textarea id="clear-event-demo" label="Textarea with clear event" name="demo" clearable value="Try to clear me!"></m-textarea>
</form>
<p id="clear-output"></p>`}
  js={`const textarea = document.querySelector('#clear-event-demo');
const output = document.querySelector('#clear-output');
textarea.addEventListener('m-textarea-clear', (e) => {
  output.textContent = 'Clear button clicked!';
});`}
/>
